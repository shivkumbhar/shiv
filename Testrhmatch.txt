SELECT
    da.host_name AS asset_name,
    dos.description AS os,
    cve.reference AS cve,
    rhsa.reference AS errata,
    dv.date_published AS published_date,
    ROUND(EXTRACT(EPOCH FROM (NOW() - dv.date_published)) / 86400) AS days_since_published
FROM fact_asset_vulnerability_finding favf
JOIN dim_asset da ON favf.asset_id = da.asset_id
JOIN dim_operating_system dos ON da.operating_system_id = dos.operating_system_id
LEFT JOIN dim_vulnerability dv ON favf.vulnerability_id = dv.vulnerability_id
LEFT JOIN dim_vulnerability_reference cve
     ON dv.vulnerability_id = cve.vulnerability_id AND cve.source = 'CVE'
LEFT JOIN dim_vulnerability_reference rhsa
     ON dv.vulnerability_id = rhsa.vulnerability_id AND lower(rhsa.source) LIKE '%redhat%'
WHERE dos.description ILIKE '%linux%' OR dos.description ILIKE '%red hat%' OR dos.description ILIKE '%rhel%'
ORDER BY days_since_published DESC
LIMIT 100;
