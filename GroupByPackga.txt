SELECT
    dsw.name AS software,
    dsw.version AS installed_version,
    string_agg(DISTINCT dvr.reference, ', ' ORDER BY dvr.reference) AS cves,
    COUNT(DISTINCT dvr.reference) AS cve_count,
    COUNT(DISTINCT favi.asset_id) AS affected_hosts
FROM fact_asset_vulnerability_instance favi
JOIN dim_vulnerability_reference dvr
     ON favi.vulnerability_id = dvr.vulnerability_id
    AND dvr.source = 'CVE'
JOIN dim_asset_software das
     ON favi.asset_id = das.asset_id
JOIN dim_software dsw
     ON das.software_id = dsw.software_id
WHERE favi.proof ILIKE '%' || dsw.name || '%'      -- proof references this installed software
GROUP BY dsw.name, dsw.version
ORDER BY cve_count DESC, software;
