SELECT
    CASE
        WHEN da.host_name ILIKE '%prod%' THEN 'PROD'
        WHEN da.host_name ILIKE '%dev%'  THEN 'DEV'
        ELSE 'OTHER'
    END AS environment,
    -- package/component derived from the check title (e.g. "Red Hat: CVE-2024-1234: openssl security update" -> openssl)
    COALESCE(
        substring(dv.title FROM 'CVE-\d{4}-\d+:\s*([a-z0-9][a-zA-Z0-9\.\+\-_]{2,})'),
        left(ds.summary, 80)
    ) AS package_to_upgrade,
    -- concrete fix instruction from the solution
    left(proofAsText(ds.fix), 150) AS fix_action,
    string_agg(DISTINCT
        CASE
            WHEN dv.nexpose_id ILIKE 'redhat-rhsa%' THEN
                'RHSA-' || substring(dv.nexpose_id FROM 'rhsa-(\d{4})-') || ':' ||
                substring(dv.nexpose_id FROM 'rhsa-\d{4}-(\d+)')
        END, ', ') AS errata,
    string_agg(DISTINCT substring(dv.title FROM '(CVE-\d{4}-\d+)'), ', ') AS cves,
    COUNT(DISTINCT dv.vulnerability_id) AS vuln_count,
    COUNT(DISTINCT da.asset_id) AS affected_hosts,
    SUM(COUNT(DISTINCT dv.vulnerability_id)) OVER (
        PARTITION BY CASE
            WHEN da.host_name ILIKE '%prod%' THEN 'PROD'
            WHEN da.host_name ILIKE '%dev%'  THEN 'DEV'
            ELSE 'OTHER'
        END
    ) AS environment_total_vulns
FROM dim_asset_vulnerability_best_solution davbs
JOIN dim_vulnerability dv USING (vulnerability_id)
JOIN dim_asset da ON davbs.asset_id = da.asset_id
JOIN dim_solution ds USING (solution_id)
GROUP BY environment, package_to_upgrade, left(proofAsText(ds.fix), 150)
ORDER BY environment, vuln_count DESC;
