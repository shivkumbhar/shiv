SELECT
    left(dv.title, 80) AS vuln_title,
    string_agg(DISTINCT dvr.reference, ', ') AS cves,
    COUNT(DISTINCT dvr.reference) AS cve_count,
    COUNT(DISTINCT favf.asset_id) AS affected_hosts
FROM fact_asset_vulnerability_finding favf
JOIN dim_vulnerability dv ON favf.vulnerability_id = dv.vulnerability_id
JOIN dim_vulnerability_reference dvr
     ON favf.vulnerability_id = dvr.vulnerability_id AND dvr.source = 'CVE'
GROUP BY vuln_title
ORDER BY cve_count DESC;
