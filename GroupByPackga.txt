SELECT
    COALESCE(
        substring(dv.title FROM ':\s*(?:CVE-\d{4}-\d+:)?\s*([a-z0-9][a-z0-9\.\+\-]{2,})'),
        left(dv.title, 60)
    ) AS package_group,
    string_agg(DISTINCT dvr.reference, ', ' ORDER BY dvr.reference) AS cves,
    COUNT(DISTINCT dvr.reference) AS cve_count,
    COUNT(DISTINCT favf.asset_id) AS affected_hosts
FROM fact_asset_vulnerability_finding favf
JOIN dim_vulnerability dv
     ON favf.vulnerability_id = dv.vulnerability_id
JOIN dim_vulnerability_reference dvr
     ON favf.vulnerability_id = dvr.vulnerability_id
    AND dvr.source = 'CVE'
GROUP BY package_group
ORDER BY cve_count DESC;
