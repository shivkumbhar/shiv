SELECT
    COALESCE(
        substring(favi.proof FROM '([a-zA-Z0-9_+\-\.]+)-[0-9][^ ]*'),
        'unparsed: ' || left(dv.title, 60)
    ) AS package,
    string_agg(DISTINCT dvr.reference, ', ' ORDER BY dvr.reference) AS cves,
    COUNT(DISTINCT dvr.reference) AS cve_count,
    COUNT(DISTINCT favi.asset_id) AS affected_hosts
FROM fact_asset_vulnerability_instance favi
JOIN dim_vulnerability dv ON favi.vulnerability_id = dv.vulnerability_id
JOIN dim_vulnerability_reference dvr
     ON dv.vulnerability_id = dvr.vulnerability_id AND dvr.source = 'CVE'
GROUP BY package
ORDER BY cve_count DESC;
